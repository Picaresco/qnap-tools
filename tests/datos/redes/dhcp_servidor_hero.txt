== dhcpd_docker0.conf
max-lease-time 43200;
default-lease-time 43200;
ddns-update-style interim;
allow booting;
allow bootp;
class "pxeclients" {
match if substring(option vendor-class-identifier,0,9)="PXEClient";
}
subnet 192.0.2.2 netmask 255.255.255.0{
interface "docker0";
range 192.0.2.3 192.0.2.4;
option routers 192.0.2.5;
option domain-name-servers 192.0.2.5, 0.0.0.0;
}
